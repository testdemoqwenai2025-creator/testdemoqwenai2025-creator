    33→    dt = NOW - timedelta(minutes=minutes_ago, seconds=seconds_offset)

   389→# Stage 2: Compliance Metrics — Swarm-specific KPIs
   390→# ══════════════════════════════════════════════════════════════════════════════

   392→def generate_metrics(num_points=60):
   393→    points = {k: [] for k in [
   394→        "ingestion_throughput", "imperative_extraction_rate", "violation_detection_rate",
   395→        "remediation_completion_rate", "boilerplate_reduction_pct", "compliance_posture_score",
   396→        "adversarial_audit_pass_rate", "human_approval_latency_hr", "token_cost_saved_usd",
   397→    ]}
   398→
   399→    base = {
   400→        "ingestion_throughput": 14,           # regulations ingested/hour
   401→        "imperative_extraction_rate": 38,     # imperatives/hour
   402→        "violation_detection_rate": 4.2,      # violations/hour
   403→        "remediation_completion_rate": 92.3,  # % of violations remediated within SLA
   404→        "boilerplate_reduction_pct": 73.5,    # PDF claim: 60-80%
   405→        "compliance_posture_score": 87.4,     # overall org posture (0-100)
   406→        "adversarial_audit_pass_rate": 44.8,  # % of audits that detect violations (adversarial stance)
   407→        "human_approval_latency_hr": 2.4,     # avg hours from detection to human approval
   408→        "token_cost_saved_usd": 127.80,       # $ saved per hour from boilerplate stripping
   409→    }
   410→
   411→    for i in range(num_points):
   412→        t = ts(minutes_ago=(num_points - i))
   413→        time_factor = math.sin(i / num_points * 2 * math.pi) * 5
   414→        burst = 8 if 18 <= i <= 23 else 0  # regulatory publication burst
   415→
   416→        for k, b in base.items():
   417→            noise = random.gauss(0, b * 0.03)
   418→            if k == "compliance_posture_score":
   419→                v = round(max(45, min(100, b - i * 0.04 + time_factor * 0.2 - burst * 0.4 + noise)), 1)
   420→            elif k == "remediation_completion_rate":
   421→                v = round(max(50, min(100, b + time_factor * 0.5 - burst * 2 + noise)), 1)
   422→            elif k == "boilerplate_reduction_pct":
   423→                v = round(max(50, min(85, b + noise)), 1)
   424→            elif k == "token_cost_saved_usd":
   425→                v = round(max(0, b * (1 + burst * 0.15) + noise), 2)
   426→            elif k == "human_approval_latency_hr":
   427→                v = round(max(0.2, b + burst * 0.3 + noise), 2)
   428→            elif "rate" in k and "pass" not in k and "completion" not in k:
   429→                v = round(max(0.1, b + burst + noise), 2)
   430→            elif k == "adversarial_audit_pass_rate":
   431→                v = round(max(20, min(80, b + time_factor * 0.4 + noise)), 1)
   432→            else:
   433→                v = round(max(0.1, b + burst + noise), 2)
   434→            points[k].append({"timestamp": t, "value": v})
   435→
   436→    metric_meta = {
   437→        "ingestion_throughput": ("regulations/hour", "Agent 1 — Federal Register polling throughput"),
   438→        "imperative_extraction_rate": ("imperatives/hour", "Agent 2 — Imperative extraction throughput"),
   439→        "violation_detection_rate": ("violations/hour", "Agent 3 — Violation detection rate"),

   881→def generate_state_machine(num_transitions=40):
   882→    """SKILLS.md §5: State Management — compliance state machine."""
   883→    entities = []
   884→    for i, (etype, edesc, jurisdiction) in enumerate(ENTITY_TYPES):
   885→        eid = f"ENT-{etype.upper()[:3]}-{i+1:03d}"
   886→        entities.append({
   887→            "id": eid,
   888→            "type": etype,
   889→            "description": edesc,
   890→            "jurisdiction": jurisdiction,
   891→            "current_state": "compliant",
   892→            "registered_at": ts(minutes_ago=240 - i * 10),
   893→            "compliance_score": 100.0,
   894→            "risk_factors": [],
   895→        })
   896→
   897→    transitions = []
   898→    state_histories = {e["id"]: [e["current_state"]] for e in entities}
   899→    metrics = {"transitions": 0, "escalations": 0, "resolutions": 0, "invalid_attempts": 0}
   900→
   901→    for i in range(num_transitions):
   902→        entity = rand_choice(entities)
   903→        from_state = entity["current_state"]
   904→        valid_targets = VALID_TRANSITIONS.get(from_state, [])
   905→        if not valid_targets:
   906→            continue
   907→        to_state = rand_choice(valid_targets)
   908→        trigger = rand_choice(TRIGGERS)
   909→        transition_id = f"TRN-{uuid.uuid4().hex[:8].upper()}"
   910→        approved_by = None
   911→        if to_state == "under_remediation" or to_state == "escalated":
   912→            approved_by = rand_choice(["cco@company.com", "ciso@company.com", "legal-counsel@company.com"])
   913→
   914→        transition = {
   915→            "transition_id": transition_id,
   916→            "entity_id": entity["id"],
   917→            "entity_type": entity["type"],
   918→            "entity_description": entity["description"],
   919→            "from_state": from_state,
   920→            "to_state": to_state,
   921→            "trigger": trigger,
   922→            "evidence": [
   923→                f"Triggered by: {trigger}",
   924→                f"Trace ID: {uuid.uuid4().hex[:16]}",
   925→                f"Regulation: {entity['jurisdiction']}",
   926→            ],
   927→            "timestamp": ts(minutes_ago=random.randint(0, 200)),
   928→            "trace_id": uuid.uuid4().hex[:16],
   929→            "approved_by": approved_by,
   930→            "compliance_score_before": entity["compliance_score"],
   931→            "compliance_score_after": round(max(0, min(100, entity["compliance_score"] + STATE_SCORE_DELTA[to_state])), 1),
   932→        }
   933→        transitions.append(transition)
   934→        entity["current_state"] = to_state
   935→        entity["compliance_score"] = transition["compliance_score_after"]
   936→        state_histories[entity["id"]].append(to_state)
   937→        metrics["transitions"] += 1
   938→        if to_state == "escalated":
   939→            metrics["escalations"] += 1
   940→        if to_state == "compliant" and from_state != "audit_pending":
   941→            metrics["resolutions"] += 1
   942→
   943→    transitions.sort(key=lambda t: t["timestamp"], reverse=True)
   944→    return {
   945→        "entities": entities,
   946→        "total_transitions": len(transitions),

   949→        "valid_transitions": VALID_TRANSITIONS,
   950→        "metrics": metrics,
   951→        "state_distribution": {
