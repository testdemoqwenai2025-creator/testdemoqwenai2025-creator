   400→# Stage 2: Compliance Metrics — Swarm-specific KPIs
   401→# ══════════════════════════════════════════════════════════════════════════════
   402→
   403→def generate_metrics(num_points=60):
   404→    points = {k: [] for k in [
   405→        "ingestion_throughput", "imperative_extraction_rate", "violation_detection_rate",
   406→        "remediation_completion_rate", "boilerplate_reduction_pct", "compliance_posture_score",
   407→        "adversarial_audit_pass_rate", "human_approval_latency_hr", "token_cost_saved_usd",
   408→    ]}
   409→
   410→    base = {
   411→        "ingestion_throughput": 14,           # regulations ingested/hour
   412→        "imperative_extraction_rate": 38,     # imperatives/hour
   413→        "violation_detection_rate": 4.2,      # violations/hour
   414→        "remediation_completion_rate": 92.3,  # % of violations remediated within SLA
   415→        "boilerplate_reduction_pct": 73.5,    # PDF claim: 60-80%
   416→        "compliance_posture_score": 87.4,     # overall org posture (0-100)
   417→        "adversarial_audit_pass_rate": 44.8,  # % of audits that detect violations (adversarial stance)
   418→        "human_approval_latency_hr": 2.4,     # avg hours from detection to human approval
   419→        "token_cost_saved_usd": 127.80,       # $ saved per hour from boilerplate stripping
   420→    }
   421→
   422→    for i in range(num_points):
   423→        t = ts(minutes_ago=(num_points - i))
   424→        time_factor = math.sin(i / num_points * 2 * math.pi) * 5
   425→        burst = 8 if 18 <= i <= 23 else 0  # regulatory publication burst
   426→
   427→        for k, b in base.items():
   428→            noise = random.gauss(0, b * 0.03)
   429→            if k == "compliance_posture_score":
   430→                v = round(max(45, min(100, b - i * 0.04 + time_factor * 0.2 - burst * 0.4 + noise)), 1)
   431→            elif k == "remediation_completion_rate":
   432→                v = round(max(50, min(100, b + time_factor * 0.5 - burst * 2 + noise)), 1)
   433→            elif k == "boilerplate_reduction_pct":
   434→                v = round(max(50, min(85, b + noise)), 1)
   435→            elif k == "token_cost_saved_usd":
   436→                v = round(max(0, b * (1 + burst * 0.15) + noise), 2)
   437→            elif k == "human_approval_latency_hr":
   438→                v = round(max(0.2, b + burst * 0.3 + noise), 2)
   439→            elif "rate" in k and "pass" not in k and "completion" not in k:
   440→                v = round(max(0.1, b + burst + noise), 2)
   441→            elif k == "adversarial_audit_pass_rate":
   442→                v = round(max(20, min(80, b + time_factor * 0.4 + noise)), 1)
   443→            else:
   444→                v = round(max(0.1, b + burst + noise), 2)
   445→            points[k].append({"timestamp": t, "value": v})

   892→def generate_state_machine(num_transitions=40):
   893→    """SKILLS.md §5: State Management — compliance state machine."""
   894→    entities = []
   895→    for i, (etype, edesc, jurisdiction) in enumerate(ENTITY_TYPES):
   896→        eid = f"ENT-{etype.upper()[:3]}-{i+1:03d}"
   897→        entities.append({
   898→            "id": eid,
   899→            "type": etype,
   900→            "description": edesc,
   901→            "jurisdiction": jurisdiction,
   902→            "current_state": "compliant",
   903→            "registered_at": ts(minutes_ago=240 - i * 10),
   904→            "compliance_score": 100.0,
   905→            "risk_factors": [],
   906→        })
   907→
   908→    transitions = []
   909→    state_histories = {e["id"]: [e["current_state"]] for e in entities}
   910→    metrics = {"transitions": 0, "escalations": 0, "resolutions": 0, "invalid_attempts": 0}
   911→
   912→    for i in range(num_transitions):
   913→        entity = rand_choice(entities)
   914→        from_state = entity["current_state"]
   915→        valid_targets = VALID_TRANSITIONS.get(from_state, [])
   916→        if not valid_targets:
   917→            continue
   918→        to_state = rand_choice(valid_targets)
   919→        trigger = rand_choice(TRIGGERS)
   920→        transition_id = f"TRN-{uuid.uuid4().hex[:8].upper()}"
   921→        approved_by = None
   922→        if to_state == "under_remediation" or to_state == "escalated":
   923→            approved_by = rand_choice(["cco@company.com", "ciso@company.com", "legal-counsel@company.com"])
   924→
   925→        transition = {
   926→            "transition_id": transition_id,
   927→            "entity_id": entity["id"],
   928→            "entity_type": entity["type"],
   929→            "entity_description": entity["description"],
   930→            "from_state": from_state,
   931→            "to_state": to_state,
   932→            "trigger": trigger,
   933→            "evidence": [
   934→                f"Triggered by: {trigger}",
   935→                f"Trace ID: {uuid.uuid4().hex[:16]}",
   936→                f"Regulation: {entity['jurisdiction']}",
   937→            ],
   938→            "timestamp": ts(minutes_ago=random.randint(0, 200)),
   939→            "trace_id": uuid.uuid4().hex[:16],
   940→            "approved_by": approved_by,
   941→            "compliance_score_before": entity["compliance_score"],
   942→            "compliance_score_after": round(max(0, min(100, entity["compliance_score"] + STATE_SCORE_DELTA[to_state])), 1),
   943→        }
   944→        transitions.append(transition)
   945→        entity["current_state"] = to_state
   946→        entity["compliance_score"] = transition["compliance_score_after"]
   947→        state_histories[entity["id"]].append(to_state)
   948→        metrics["transitions"] += 1
   949→        if to_state == "escalated":
   950→            metrics["escalations"] += 1
   951→        if to_state == "compliant" and from_state != "audit_pending":
   952→            metrics["resolutions"] += 1
   953→
   954→    transitions.sort(key=lambda t: t["timestamp"], reverse=True)
   955→    return {
   956→        "entities": entities,
   957→        "total_transitions": len(transitions),
   958→        "transitions": transitions[:30],  # Top 30 most recent
   959→        "state_histories": state_histories,
   960→        "valid_transitions": VALID_TRANSITIONS,
   961→        "metrics": metrics,
   962→        "state_distribution": {
   963→            state: len([e for e in entities if e["current_state"] == state])
   964→            for state in COMPLIANCE_STATES
   965→        },
   966→    }

  1086→            "status": "resolved" if is_resolved else "pending",
  1087→            "resolution_strategy": strategy if is_resolved else None,
  1088→            "resolution_description": strategy_desc if is_resolved else None,
  1089→            "resolution": f"Resolved using strategy: {strategy}" if is_resolved else None,
  1090→            "resolved_at": ts(minutes_ago=random.randint(0, 100)) if is_resolved else None,
  1091→            "detected_at": ts(minutes_ago=random.randint(0, 200)),
  1092→            "winner": "regulation_a" if is_resolved and strategy in ("regulation_a_takes_precedence", "apply_strictest") and reg_a["tier"] == "Critical"
  1093→                     else "regulation_b" if is_resolved and strategy in ("regulation_a_takes_precedence", "apply_strictest") and reg_b["tier"] == "Critical"
  1094→                     else ("merged" if is_resolved and strategy == "merge_requirements"

  1117→    templates = {

  1121→        "requirement_contradiction": f"Direct contradiction: '{clause_a}' (from {reg_a['name']}) vs '{clause_b}' (from {reg_b['name']})",
  1122→        "scope_overlap": f"Both {reg_a['name']} and {reg_b['name']} apply to same scope — '{clause_a}' vs '{clause_b}'",

  1127→def generate_audit_chain(num_entries=30):
  1128→    """SKILLS.md §5: Immutability & Versioning — append-only audit chain with hash linking."""
  1129→    entries = []
  1130→    prev_hash = "0" * 64  # Genesis block
  1131→    
  1132→    for i in range(num_entries):
  1133→        event_type = rand_choice([
  1134→            "state_transition", "agent_output_published", "violation_detected",
  1135→            "remediation_artifact_generated", "human_approval_granted", "human_approval_denied",
  1136→            "conflict_detected", "conflict_resolved", "schema_validation_passed",
  1137→            "audit_trail_signed", "evidence_collected",
  1138→        ])
  1139→        agent = rand_choice(AGENTS)["name"]
  1140→        entity_id = f"ENT-{rand_choice(['PHI','PII','API','IAM','AI','INF'])}-{random.randint(1,8):03d}"
  1141→        timestamp = ts(minutes_ago=(num_entries - i))
  1142→        
  1143→        payload = {
  1144→            "event_type": event_type,
  1145→            "agent": agent,
  1146→            "entity_id": entity_id,
  1147→            "trace_id": uuid.uuid4().hex[:16],
  1148→            "scenario_id": f"SCN-{uuid.uuid4().hex[:8].upper()}" if random.random() < 0.5 else None,

  1150→            "details": f"{event_type.replace('_', ' ')} on {entity_id}",
  1151→        }
  1152→        payload_str = json.dumps(payload, sort_keys=True)
  1153→        current_hash = hashlib.sha256((prev_hash + payload_str).encode()).hexdigest()
  1154→        verification_status = "verified" if random.random() > 0.05 else "mismatch"  # 5% tampering simulation
  1155→        
  1156→        entries.append({

  1159→            "timestamp": timestamp,
  1160→            "event_type": event_type,
  1161→            "agent": agent,
  1162→            "entity_id": entity_id,
  1163→            "payload": payload,
  1164→            "previous_hash": prev_hash[:32] + "..." if len(prev_hash) > 32 else prev_hash,
  1165→            "current_hash": current_hash[:32] + "..." if len(current_hash) > 32 else current_hash,
  1166→            "full_previous_hash": prev_hash,

  1171→            "signature_algorithm": "SHA-256 + RSA-4096" if verification_status == "verified" else None,
  1172→        })
  1173→        prev_hash = current_hash
  1174→    
  1175→    verified_count = len([e for e in entries if e["verification_status"] == "verified"])
  1176→    mismatch_count = len(entries) - verified_count
  1177→    return {
  1178→        "entries": entries,
  1179→        "total_entries": len(entries),
  1180→        "verified_entries": verified_count,
  1181→        "mismatch_entries": mismatch_count,
  1182→        "chain_integrity": "intact" if mismatch_count == 0 else "compromised",
  1183→        "genesis_hash": "0" * 64,
  1184→        "latest_hash": entries[-1]["full_current_hash"] if entries else None,
  1185→        "storage_backend": "S3 + DynamoDB (append-only, versioned)",
  1186→        "retention_policy": "7 years (regulatory requirement)",
  1187→        "signature_algorithm": "SHA-256 + RSA-4096 (HSM-backed)",
