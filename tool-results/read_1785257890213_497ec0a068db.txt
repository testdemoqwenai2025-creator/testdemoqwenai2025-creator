   883→   883→    entities = []

   900→   900→

   905→   905→        if not valid_targets:
   906→   906→            continue
   907→   907→        to_state = rand_choice(valid_targets)
   908→   908→        trigger = rand_choice(TRIGGERS)
   909→   909→        transition_id = f"TRN-{uuid.uuid4().hex[:8].upper()}"
   910→   910→        approved_by = None
   911→   911→        if to_state == "under_remediation" or to_state == "escalated":
   912→   912→            approved_by = rand_choice(["cco@company.com", "ciso@company.com", "legal-counsel@company.com"])
   913→   913→
   914→   914→        transition = {
   915→   915→            "transition_id": transition_id,
   916→   916→            "entity_id": entity["id"],
   917→   917→            "entity_type": entity["type"],
   918→   918→            "entity_description": entity["description"],
   919→   919→            "from_state": from_state,
   920→   920→            "to_state": to_state,
   921→   921→            "trigger": trigger,

   930→   930→            "compliance_score_before": entity["compliance_score"],
   931→   931→            "compliance_score_after": round(max(0, min(100, entity["compliance_score"] + STATE_SCORE_DELTA[to_state])), 1),
   932→   932→        }
   933→   933→        transitions.append(transition)
   934→   934→        entity["current_state"] = to_state
   935→   935→        entity["compliance_score"] = transition["compliance_score_after"]
   936→   936→        state_histories[entity["id"]].append(to_state)
   937→   937→        metrics["transitions"] += 1
   938→   938→        if to_state == "escalated":
   939→   939→            metrics["escalations"] += 1

   941→   941→            metrics["resolutions"] += 1
   942→   942→
   943→   943→    transitions.sort(key=lambda t: t["timestamp"], reverse=True)
   944→   944→    return {
   945→   945→        "entities": entities,
   946→   946→        "total_transitions": len(transitions),

   948→   948→        "state_histories": state_histories,
   949→   949→        "valid_transitions": VALID_TRANSITIONS,
   950→   950→        "metrics": metrics,
   951→   951→        "state_distribution": {
   952→   952→            state: len([e for e in entities if e["current_state"] == state])
   953→   953→            for state in COMPLIANCE_STATES
   954→   954→        },
   955→   955→    }
