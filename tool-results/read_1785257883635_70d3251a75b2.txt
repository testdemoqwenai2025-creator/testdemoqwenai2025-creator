   131→    # ============================================================
   132→    # COMPONENT 2: Immutable Audit Logging Engine
   133→    # ============================================================
   134→    def log_event(self, event_type, details):
   135→        timestamp = datetime.datetime.now().isoformat()
   136→        seq = len(self.audit_log) + 1
   137→        log_entry = f"{timestamp} | SEQ:{seq:04d} | {event_type} | {details}"
   138→        log_hash = hashlib.sha256(log_entry.encode()).hexdigest()
   139→        prev_hash = self.audit_log[-1]["entry_hash"] if self.audit_log else "GENESIS"
   140→        chain_hash = hashlib.sha256(f"{prev_hash}{log_hash}".encode()).hexdigest()
   141→        record = {
   142→            "seq": seq, "timestamp": timestamp, "event_type": event_type,
   143→            "details": details, "entry_hash": log_hash, "chain_hash": chain_hash,
   144→            "prev_chain_hash": prev_hash
   145→        }
   146→        self.audit_log.append(record)
   147→        return record

   219→    def detokenize_record(self, tokenized_record, token_map):
   220→        restored = copy.deepcopy(tokenized_record)
   221→        for field in ["name", "mrn", "email", "phone", "ssn"]:
   222→            if field in restored and isinstance(restored[field], str) and restored[field].startswith("TKN-"):
   223→                if restored[field] in token_map:
   224→                    restored[field] = token_map[restored[field]]
   225→        restored.pop("_tokenization_meta", None)
   226→        entry = {
   227→            "component": 5, "name": "Tokenization Engine (FHIR)",
   228→            "event": "DETOKENIZE", "status": "success"
   229→        }
   230→        self.log_event("DETOKENIZE", json.dumps({k: v for k, v in entry.items() if k not in ["component", "name"]}))
   231→        return restored, entry

   267→    # COMPONENT 7: Retention Policy Enforcer
   268→    # ============================================================
   269→    def enforce_retention(self, data_category, created_date_str, current_date_str=None):
   270→        created = datetime.datetime.fromisoformat(created_date_str)
   271→        current = datetime.datetime.fromisoformat(current_date_str) if current_date_str else datetime.datetime.now()
   272→        policy = self.retention_policies.get(data_category, {"max_age_days": 365, "action": "review"})
   273→        age_days = (current - created).days
   274→        max_age = policy["max_age_days"]
   275→        status = "active" if age_days < max_age * 0.8 else ("warning" if age_days < max_age else "expired")
   276→        action_taken = policy["action"] if status == "expired" else "none"
   277→        entry = {
   278→            "component": 7, "name": "Retention Policy Enforcer",
   279→            "event": "RETENTION_CHECK", "data_category": data_category,
   280→            "age_days": age_days, "max_age_days": max_age,
   281→            "utilization_pct": round(age_days / max_age * 100, 1),
   282→            "status": status, "action_taken": action_taken
   283→        }
   284→        self.log_event("RETENTION_CHECK", json.dumps({k: v for k, v in entry.items() if k not in ["component", "name"]}))
   285→        return {"status": status, "action_taken": action_taken, "utilization_pct": entry["utilization_pct"]}, entry

   287→    # ============================================================
   288→    # COMPONENT 8: Data Classification & Sensitivity Labeller
   289→    # ============================================================
   290→    def classify_data(self, data_content):
   291→        labels = []
   292→        content_lower = data_content.lower()
   293→        for label, keywords in self.classification_rules.items():
   294→            for kw in keywords:
   295→                if kw.lower() in content_lower:
   296→                    labels.append(label)
   297→                    break
   298→        if not labels:
   299→            labels.append("Unclassified")
   300→        sensitivity_score = sum([3 if l == "PHI" else 2 if l == "Restricted" else 1 for l in labels])
   301→        entry = {
   302→            "component": 8, "name": "Data Classification & Sensitivity Labeller",
   303→            "event": "CLASSIFY", "labels_assigned": labels,
   304→            "sensitivity_score": sensitivity_score, "max_possible": 9,
   305→            "risk_level": "Critical" if sensitivity_score >= 5 else ("High" if sensitivity_score >= 3 else "Medium")
   306→        }
   307→        self.log_event("CLASSIFY", json.dumps({k: v for k, v in entry.items() if k not in ["component", "name"]}))
   308→        return labels, sensitivity_score, entry
